// Include RadioHead Amplitude Shift Keying Library

#include <RH_ASK.h>

// Include dependant SPI Library

#include <SPI.h> 

// Create Amplitude Shift Keying Object

RH_ASK rf_driver;

 

const int MQ3_PIN = A0;

const int digitalInPin = 2;

const int BUZZER_LED_PIN = 7;

const int BUZZER_PIN = 8;

const int ALOCHOL_THRESHOLD = 500;

int Helmetvalue = 0;

 

void setup()

{

 

  Serial.begin(9600);

 //print (HELMET_SWITCH, INPUT);

 

 pinMode(BUZZER_LED_PIN, OUTPUT);

 pinMode(BUZZER_PIN,OUTPUT);

 

 // Initialize ASK Object

 rf_driver.init();

}

void loop()

{//Buzzer 5 times beep

  for (int i=0; i<5; i++)

  {

    int alcoholValue  = analogRead(MQ3_PIN);    //(alcohole detect= yes if value of alcohol is greater than 500)

   

    Helmetvalue= digitalRead(digitalInPin);     //(0= driver wear helmet = yes)

   

    if (Helmetvalue==0) 

    {

      if(ALOCHOL_THRESHOLD < alcoholValue)    //helmet on alcohol on

      {

        Serial.write('U');

        digitalWrite(BUZZER_LED_PIN,HIGH);

        digitalWrite(BUZZER_PIN,HIGH);

 

        const char *msg = "HELMET OK Alcohol Drink";

        rf_driver.send((uint8_t *)msg, strlen(msg));

        rf_driver.waitPacketSent();

        delay(1000);

      }

 

      else if(ALOCHOL_THRESHOLD > alcoholValue) //helmet on alcohol off

      {

        Serial.write('S');

        digitalWrite(BUZZER_LED_PIN,LOW);

        digitalWrite(BUZZER_PIN,LOW);

 

        const char *msg = "HELMET OK No Alcohol Drink";

        rf_driver.send((uint8_t *)msg, strlen(msg));

        rf_driver.waitPacketSent();

        delay(1000);

      }

    }

   

    else

    {

      Serial.write('U');

      digitalWrite(BUZZER_LED_PIN,LOW);

      //{

      // digitalwrite (BUZZER_PIN,HIGH);

      delay(500);

      //digitalwrite(BUZZER_PIN,LOW);

      delay(500);

      // }

 

      //After 5 times, buzzer LOW

      digitalWrite(BUZZER_PIN,HIGH);

 

      const char *msg = "HELMET_NO";

      rf_driver.send((uint8_t *)msg, strlen(msg));

      rf_driver.waitPacketSent();

      delay(1000);

 

    }

   

    Serial.println(Helmetvalue);

    Serial.println(alcoholValue);

    delay(100);

  }

}

#include <RH_ASK.h>

#include <SPI.h>

#include <string.h>

 

// Create Amplitude Shift Keying Object

RH_ASK rf_driver;

//RH_ASK driver(2000, 11, 12, 10);

 

const int relayPin = 4;

const int ledPin = 5;

const int buzzerPin = 6;

 

unsigned long lastSignal = 0;

bool helmetOK = false;

 

void setup() 

{

  // Initialize ASK Object

  rf_driver.init();

  // Setup Serial Monitor

  Serial.begin(9600);

 

  pinMode(relayPin, OUTPUT);

  pinMode(ledPin, OUTPUT);

  pinMode(buzzerPin, OUTPUT);

 

  digitalWrite(relayPin, LOW);

  digitalWrite(ledPin, LOW);

  digitalWrite(buzzerPin, LOW);

 

  //driver.init();

}

 

void loop() 

{

  uint8_t buf[27];

  uint8_t buflen = sizeof(buf);

 

  if (rf_driver.recv(buf, &buflen)) 

  {

    if (buflen < sizeof(buf))

    {

      buf[buflen] = '\0';

 

      if (strcmp((char*)buf, "HELMET OK No Alcohol Drink") == 0) 

      {

        helmetOK = true;

        lastSignal = millis();

        Serial.write("HELMET_OK_no_alcohole");

        Serial.write("MOTOR_ON");

      }

      else if (strcmp((char*)buf, "HELMET OK Alcohol Drink") == 0) 

      {

        helmetOK = false;

        lastSignal = millis();

        Serial.write("alcohol_drink");

        Serial.write("MOTOR_OFF");

      }

    }

  }

 

  // Signal बंद झाल्यास Relay OFF

  if (millis() - lastSignal > 1000) 

  {

    helmetOK = false;

  }

 

  if (helmetOK) 

  {

    digitalWrite(relayPin, HIGH);

    digitalWrite(ledPin, LOW);

    digitalWrite(buzzerPin, LOW);

    Serial.write("HELMET_OK_no_alcohole");

    Serial.write("MOTOR_ON");

  } 

  else 

  {

    digitalWrite(relayPin, LOW);

    digitalWrite(ledPin, HIGH);

    digitalWrite(buzzerPin, HIGH);

    Serial.write("alcohol_drink");

    Serial.write("MOTOR_OFF");

  }

}
